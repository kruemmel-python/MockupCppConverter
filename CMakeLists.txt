cmake_minimum_required(VERSION 3.20)
project(MockupCppConverter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(MockupCppConverter WIN32 src/main.cpp)
target_compile_definitions(MockupCppConverter PRIVATE UNICODE _UNICODE)
target_link_libraries(MockupCppConverter PRIVATE comdlg32 shell32 ole32)
if(MINGW)
  target_link_options(MockupCppConverter PRIVATE -static -static-libgcc -static-libstdc++)
endif()
